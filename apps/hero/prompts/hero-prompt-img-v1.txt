(function () {
  'use strict';

  /* ---------- Settings ---------- */
  var DEFAULT_COLOURS = {
    light: { background: 'slate-50',  h1: 'slate-900', h2: 'slate-700' },
    dark:  { background: 'slate-900', h1: 'slate-50',  h2: 'slate-300' }
  };

  var settings = {
    dark: true,
    centre: true,
    minHeight: true,
    h1Serif: false,
    h2Serif: false,
    h2Hidden: false,
    h1Text: 'Lorem ipsum dolor sit amet',
    h2Text: 'Consectetur adipiscing elit. Integer vitae libero ac risus egestas placerat.',
    colours: Object.assign({}, DEFAULT_COLOURS.dark),   // colour names, e.g. "slate-50"
    colourTarget: 'background',                         // background | start | end | h1 | h2
    gradient: {
      enabled: false,
      initialised: false,        // true once start/end have been seeded
      start: 'slate-900',
      end: 'blue-500',
      direction: 'to right'
    }
  };

  var DIRECTIONS = ['to right', 'to left', 'to bottom', 'to top'];
  var TARGET_LABELS = { background: 'Background', start: 'Start colour', end: 'End colour', h1: 'H1', h2: 'H2' };
  var STATUS_LABELS = { background: 'Background colour', start: 'Start colour', end: 'End colour', h1: 'H1 colour', h2: 'H2 colour' };
  var FAMILIES = ['slate','gray','zinc','neutral','stone','red','orange','amber','yellow','lime','green','emerald','teal','cyan','sky','blue','indigo','violet','purple','fuchsia','pink','rose'];
  var SHADES = [50,100,200,300,400,500,600,700,800,900,950];

  var heroEl = document.getElementById('hero');
  var heroH1 = document.getElementById('hero-h1');
  var heroH2 = document.getElementById('hero-h2');

  /* ---------- Hero ---------- */
  function heroClassName() {
    var c = ['hero', settings.dark ? 'hero-dark' : 'hero-light'];
    if (settings.centre) c.push('hero-centre');
    if (settings.minHeight) c.push('hero-min-height');
    if (settings.h1Serif) c.push('hero-h1-serif');
    if (settings.h2Serif) c.push('hero-h2-serif');
    return c.join(' ');
  }

  function applyHero() {
    heroEl.className = heroClassName();
    heroH1.textContent = settings.h1Text;
    heroH2.textContent = settings.h2Text;
    heroH2.hidden = settings.h2Hidden;
  }

  /* ---------- Colours ---------- */
  function colourVar(name) { return 'var(--color-' + name + ')'; }

  function getTargetColour(t) {
    if (t === 'start') return settings.gradient.start;
    if (t === 'end') return settings.gradient.end;
    return settings.colours[t];
  }

  function setTargetColour(t, name) {
    if (t === 'start') settings.gradient.start = name;
    else if (t === 'end') settings.gradient.end = name;
    else settings.colours[t] = name;
  }

  function isTargetVisible(t) {
    if (t === 'background') return !settings.gradient.enabled;
    if (t === 'start' || t === 'end') return settings.gradient.enabled;
    return true;
  }

  function isTargetEnabled(t) {
    return isTargetVisible(t) && !(t === 'h2' && settings.h2Hidden);
  }

  function ensureValidTarget() {
    if (!isTargetEnabled(settings.colourTarget)) {
      settings.colourTarget = settings.gradient.enabled ? 'start' : 'background';
    }
  }

  function buildColourCss() {
    var c = settings.colours, g = settings.gradient;
    var dir = DIRECTIONS.indexOf(g.direction) >= 0 ? g.direction : 'to right';
    var bg = g.enabled
      ? 'html .hero { background-color: ' + colourVar(g.start) + '; background-image: linear-gradient(' + dir + ', ' + colourVar(g.start) + ', ' + colourVar(g.end) + '); }'
      : 'html .hero { background-color: ' + colourVar(c.background) + '; background-image: none; }';
    return [
      bg,
      'html .hero .hero-content h1 { color: ' + colourVar(c.h1) + '; }',
      'html .hero .hero-content h2 { color: ' + colourVar(c.h2) + '; }'
    ].join('\n');
  }

  function applyColours() {
    document.getElementById('hero-colour-styles').textContent = buildColourCss();
  }

  function colourLabel(name) {
    var m = /^([a-z]+)-(\d+)$/.exec(name);
    var base = m ? m[1] + ' ' + m[2] : name;
    return base.charAt(0).toUpperCase() + base.slice(1);
  }

  /* ---------- Palette ---------- */
  var swatchGrid = [];   // rows of buttons, for keyboard navigation
  var allSwatches = [];

  function buildSwatchCss(names) {
    document.getElementById('swatch-styles').textContent = names.map(function (n) {
      return '.sw-' + n + ' { background-color: var(--color-' + n + '); }';
    }).join('\n');
  }

  function makeSwatch(name, row, col) {
    var b = document.createElement('button');
    b.type = 'button';
    b.className = 'swatch sw-' + name;
    b.dataset.colour = name;
    b.dataset.row = row;
    b.dataset.col = col;
    b.setAttribute('aria-label', colourLabel(name));
    b.setAttribute('aria-pressed', 'false');
    b.title = colourLabel(name);
    b.tabIndex = -1;
    allSwatches.push(b);
    return b;
  }

  function renderPalette() {
    var table = document.getElementById('palette');
    var names = [];
    var thead = document.createElement('thead');
    var hr = document.createElement('tr');
    var corner = document.createElement('th');
    corner.scope = 'col';
    corner.textContent = 'Colour';
    hr.appendChild(corner);
    SHADES.forEach(function (s) {
      var th = document.createElement('th');
      th.scope = 'col';
      th.textContent = s;
      hr.appendChild(th);
    });
    thead.appendChild(hr);
    table.appendChild(thead);

    var tbody = document.createElement('tbody');
    var rowIndex = 0;

    FAMILIES.forEach(function (fam) {
      var tr = document.createElement('tr');
      var th = document.createElement('th');
      th.scope = 'row';
      th.textContent = fam;
      tr.appendChild(th);
      var row = [];
      SHADES.forEach(function (s, ci) {
        var name = fam + '-' + s;
        names.push(name);
        var td = document.createElement('td');
        var b = makeSwatch(name, rowIndex, ci);
        td.appendChild(b);
        tr.appendChild(td);
        row.push(b);
      });
      swatchGrid.push(row);
      tbody.appendChild(tr);
      rowIndex++;
    });

    ['black', 'white'].forEach(function (name) {
      names.push(name);
      var tr = document.createElement('tr');
      var th = document.createElement('th');
      th.scope = 'row';
      th.textContent = name;
      tr.appendChild(th);
      var td = document.createElement('td');
      var b = makeSwatch(name, rowIndex, 0);
      td.appendChild(b);
      tr.appendChild(td);
      for (var i = 1; i < SHADES.length; i++) tr.appendChild(document.createElement('td'));
      swatchGrid.push([b]);
      tbody.appendChild(tr);
      rowIndex++;
    });

    table.appendChild(tbody);
    buildSwatchCss(names);

    table.addEventListener('click', function (e) {
      var b = e.target.closest('.swatch');
      if (b) setColour(b.dataset.colour);
    });

    table.addEventListener('focusin', function (e) {
      var b = e.target.closest('.swatch');
      if (!b) return;
      allSwatches.forEach(function (s) { s.tabIndex = -1; });
      b.tabIndex = 0;
    });

    table.addEventListener('keydown', function (e) {
      var b = e.target.closest('.swatch');
      if (!b) return;
      var r = +b.dataset.row, c = +b.dataset.col, nr = r, nc = c;
      switch (e.key) {
        case 'ArrowRight': nc = c + 1; break;
        case 'ArrowLeft':  nc = c - 1; break;
        case 'ArrowDown':  nr = r + 1; break;
        case 'ArrowUp':    nr = r - 1; break;
        case 'Home':       nc = 0; break;
        case 'End':        nc = Infinity; break;
        default: return;
      }
      e.preventDefault();
      if (nr < 0 || nr >= swatchGrid.length) return;
      var row = swatchGrid[nr];
      nc = Math.max(0, Math.min(nc, row.length - 1));
      row[nc].focus();
    });
  }

  function updatePalette() {
    var selected = getTargetColour(settings.colourTarget);
    allSwatches.forEach(function (b) {
      var on = b.dataset.colour === selected;
      b.setAttribute('aria-pressed', on ? 'true' : 'false');
      b.tabIndex = on ? 0 : -1;
    });
    if (!allSwatches.some(function (b) { return b.tabIndex === 0; })) allSwatches[0].tabIndex = 0;
  }

  /* ---------- Targets ---------- */

    Array.prototype.forEach.call(buttons, function (btn) {
      var t = btn.dataset.target;
      var name = getTargetColour(t);
      var disabled = t === 'h2' && settings.h2Hidden;
      btn.hidden = !isTargetVisible(t);
      btn.disabled = disabled;
      btn.setAttribute('aria-pressed', t === settings.colourTarget ? 'true' : 'false');
      btn.innerHTML = '';
      var chip = document.createElement('span');
      chip.className = 'chip sw-' + name;
      var tl = document.createElement('span');
      tl.className = 'tl';
      tl.textContent = TARGET_LABELS[t];
      var tc = document.createElement('span');
      tc.className = 'tc';
      tc.textContent = disabled ? name + ' (H2 hidden)' : name;
      btn.appendChild(chip); btn.appendChild(tl); btn.appendChild(tc);
    });
    var status = document.getElementById('colour-status');
    status.innerHTML = '';
    status.appendChild(document.createTextNode('Editing ' + STATUS_LABELS[settings.colourTarget] + ': '));
    var code = document.createElement('code');
    code.textContent = getTargetColour(settings.colourTarget);
    status.appendChild(code);
  }

  function setColour(name) {
    setTargetColour(settings.colourTarget, name);
    refreshColourUi();
  }

  function setTarget(t) {
    if (!isTargetEnabled(t)) return;
    settings.colourTarget = t;
    refreshColourUi();
  }

  function refreshColourUi() {
    ensureValidTarget();
    document.getElementById('dir-field').hidden = !settings.gradient.enabled;
    applyColours();
    renderTargets();
    updatePalette();
    updateExport();
  }

  /* ---------- Export ---------- */
  function esc(s) {
    return String(s).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
  }

  function buildExportHtml() {
    var heroCss = document.getElementById('hero-styles').textContent.replace(/\s+$/, '');
    var extraCss = document.getElementById('hero-extra-styles').textContent.replace(/\s+$/, '');
    var colourCss = buildColourCss();
    return [
      '<!DOCTYPE html>',
      '<html lang="en">',
      '<head>',
      '  <meta charset="UTF-8">',
      '  <meta name="viewport" content="width=device-width, initial-scale=1.0">',
      '  <meta name="description" content="Short, unique description of web page. Often used as the snippet shown in search results. About 120 characters is ideal.">',
      '  <title>My Hero Page</title>',
      '',
      '  <script src="https://cdn.jsdelivr.net/npm/@tailwindcss/browser@4"><\/script>',
      '',
      '  <style type="text/tailwindcss">',
      heroCss,
      '  </style>',
      '',
      '  <style>',
      extraCss,
      '  </style>',
      '',
      '  <style>',
      colourCss,
      '  </style>',
      '</head>',
      '<body>',
      '',
      '  <header class="' + heroClassName() + '">',
      '    <div class="hero-content">',
      '      <h1>' + esc(settings.h1Text) + '</h1>',
      '      <h2' + (settings.h2Hidden ? ' hidden' : '') + '>' + esc(settings.h2Text) + '</h2>',
      '    </div>',
      '  </header>',
      '',
      '</body>',
      '</html>'
    ].join('\n');
  }

  function updateExport() {
    document.getElementById('export-out').value = buildExportHtml();
  }

  /* ---------- Wiring ---------- */
  function onDarkChange(checked) {
    settings.dark = checked;
    settings.colours = Object.assign({}, DEFAULT_COLOURS[checked ? 'dark' : 'light']);
    applyHero();
    refreshColourUi();
  }

  function onShowH2Change(show) {
    settings.h2Hidden = !show;
    ensureValidTarget();
    applyHero();
    refreshColourUi();
  }

  function onGradientChange(on) {
    var g = settings.gradient;
    if (on && !g.initialised) {
      g.start = settings.colours.background;   // current solid colour
      g.end = 'blue-500';
      g.initialised = true;
    }
    g.enabled = on;
    refreshColourUi();
  }

  function onDirectionChange(value) {
    if (DIRECTIONS.indexOf(value) < 0) return;
    settings.gradient.direction = value;
    refreshColourUi();
  }

  function bindCheck(id, fn) {
    document.getElementById(id).addEventListener('change', function (e) { fn(e.target.checked); });
  }

  bindCheck('opt-dark', onDarkChange);
  bindCheck('opt-centre',    function (v) { settings.centre = v;    applyHero(); updateExport(); });
  bindCheck('opt-minheight', function (v) { settings.minHeight = v; applyHero(); updateExport(); });
  bindCheck('opt-h1serif',   function (v) { settings.h1Serif = v;   applyHero(); updateExport(); });
  bindCheck('opt-h2serif',   function (v) { settings.h2Serif = v;   applyHero(); updateExport(); });
  bindCheck('opt-showh2', onShowH2Change);
  bindCheck('opt-gradient', onGradientChange);
  document.getElementById('opt-direction').addEventListener('change', function (e) {
    onDirectionChange(e.target.value);
  });

  document.getElementById('txt-h1').addEventListener('input', function (e) {
    settings.h1Text = e.target.value; applyHero(); updateExport();
  });
  document.getElementById('txt-h2').addEventListener('input', function (e) {
    settings.h2Text = e.target.value; applyHero(); updateExport();
  });

  document.getElementById('targets').addEventListener('click', function (e) {
    var btn = e.target.closest('.target-btn');
    if (btn) setTarget(btn.dataset.target);
  });

  document.getElementById('copy-btn').addEventListener('click', function () {
    var out = document.getElementById('export-out');
    out.select();
    if (navigator.clipboard) navigator.clipboard.writeText(out.value);
    else document.execCommand('copy');
  });

  /* ---------- Init ---------- */
  document.getElementById('txt-h1').value = settings.h1Text;
  document.getElementById('txt-h2').value = settings.h2Text;
  renderPalette();
  applyHero();
  refreshColourUi();
})();
